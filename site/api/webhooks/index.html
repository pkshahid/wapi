    Webhooks¶

    wapi.webhooks.events ¶

    Typed webhook events produced by :func:wapi.webhooks.parse.

    Every event carries the field that produced it, the WABA id (entry id), and the raw value dict. Subclasses add typed accessors for known fields. Pattern-match on them::

    for event in parse(body):
    match event:
        case InboundMessages() as m:
            for msg in m.messages: ...
        case StatusUpdates() as s:
            for st in s.statuses: ...

    def parse(payload: dict[str, Any] | str | bytes) -> list[ev.WebhookEvent]:
    """Parse a webhook body into one typed event per change.

    A ``messages`` change produces an :class:`InboundMessages` event when it
    contains ``messages`` and a :class:`StatusUpdates` event when it contains
    ``statuses`` (they may both come from the same change).
    """
    if isinstance(payload, (str, bytes)):
        payload = json.loads(payload)
    body = WebhookPayload.model_validate(payload)
    out: list[ev.WebhookEvent] = []
    for entry in body.entry:
        for change in entry.changes:
            out.extend(parse_change(entry.id, change.field, change.value))
    return out

    def parse_change(waba_id: str, field: str, value: dict[str, Any]) -> list[ev.WebhookEvent]:
    """Dispatch a single ``change`` into events."""
    if field in ("messages", "group_text_messages"):
        return _parse_messages(waba_id, field, value)
    cls = _FIELD_MAP.get(field, ev.UnknownEvent)
    return [cls(field=field, waba_id=waba_id, value=value)]

    def _parse_messages(waba_id: str, field: str, value: dict[str, Any]) -> list[ev.WebhookEvent]:
    out: list[ev.WebhookEvent] = []
    metadata = None
    if isinstance(value.get("metadata"), dict):
        metadata = WebhookMetadata.model_validate(value["metadata"])

    if value.get("messages"):
        messages = [InboundMessage.model_validate(m) for m in value["messages"]]
        contacts = [WebhookContact.model_validate(c) for c in value.get("contacts") or []]
        out.append(
            ev.InboundMessages(
                field=field,
                waba_id=waba_id,
                value=value,
                messages=messages,
                contacts=contacts,
                metadata=metadata,
            )
        )
    if value.get("statuses"):
        statuses = [MessageStatus.model_validate(s) for s in value["statuses"]]
        out.append(
            ev.StatusUpdates(
                field=field,
                waba_id=waba_id,
                value=value,
                statuses=statuses,
                metadata=metadata,
            )
        )
    if value.get("message_echoes"):
        out.append(
            ev.SmbMessageEcho(
                field=field,
                waba_id=waba_id,
                value=value,
                message_echoes=list(value["message_echoes"]),
            )
        )
    if value.get("calls"):
        out.append(
            ev.CallEvent(
                field=field,
                waba_id=waba_id,
                value=value,
                calls=list(value["calls"]),
                metadata=metadata,
            )
        )
    if not out:
        out.append(ev.UnknownEvent(field=field, waba_id=waba_id, value=value))
    return out

    def compute_signature(payload: bytes, app_secret: str) -> str:
    """Return the ``sha256=…`` signature string for a payload."""
    digest = hmac.new(app_secret.encode(), payload, hashlib.sha256).hexdigest()
    return f"sha256={digest}"

    def parse(payload: dict[str, Any] | str | bytes) -> list[ev.WebhookEvent]:
    """Parse a webhook body into one typed event per change.

    A ``messages`` change produces an :class:`InboundMessages` event when it
    contains ``messages`` and a :class:`StatusUpdates` event when it contains

    if isinstance(payload, (str, bytes)):
        payload = json.loads(payload)
    body = WebhookPayload.model_validate(payload)
    out: list[ev.WebhookEvent] = []
    for entry in body.entry:
        for change in entry.changes:

    def webhook_router(
    handler: Handler | BatchHandler,
    *,
    verify_token: str,
    app_secret: str | None = None,
    path: str = "/webhook",
    per_event: bool = True,
) -> Any:
    """Build a FastAPI ``APIRouter`` handling webhook verification + dispatch.

    - ``GET {path}`` answers Meta's verification challenge.
    - ``POST {path}`` verifies ``X-Hub-Signature-256`` (when ``app_secret``
      is set), parses events, and calls ``handler``.

    With ``per_event=True`` (default) ``handler`` is invoked once per event;
    with ``False`` it receives the whole ``list[WebhookEvent]``.
    Sync and async handlers are both supported.
    """
    if not _HAS_FASTAPI:  # pragma: no cover
        raise ImportError("webhook_router requires the 'fastapi' extra: pip install wapi[fastapi]")

    router = APIRouter()

    async def _dispatch(events: list[WebhookEvent]) -> None:
        if per_event:
            for event in events:
                result = handler(event)  # type: ignore[arg-type]
                if inspect.isawaitable(result):
                    await result
        else:
            result = handler(events)  # type: ignore[arg-type]
            if inspect.isawaitable(result):
                await result

    @router.get(path)
    async def verify(request: Request) -> Response:
        challenge = verify_webhook_challenge(
            request.query_params.get("hub.mode"),
            request.query_params.get("hub.verify_token"),
            request.query_params.get("hub.challenge"),
            verify_token,
        )
        if challenge is not None:
            return PlainTextResponse(challenge)
        return PlainTextResponse("Forbidden", status_code=403)

    @router.post(path)
    async def receive(request: Request) -> Response:
        raw = await request.body()
        if app_secret is not None:
            sig = request.headers.get("x-hub-signature-256")
            if not verify_signature(raw, sig, app_secret):
                return PlainTextResponse("Invalid signature", status_code=401)
        try:
            events = parse(json.loads(raw))
        except (json.JSONDecodeError, ValueError) as e:
            raise WebhookSignatureError("malformed webhook payload") from e
        await _dispatch(events)
        return PlainTextResponse("OK")

    return router
