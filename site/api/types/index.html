    class CallResult(WapiModel):
    """Response of ``POST /{phone-number-id}/calls``."""

    messaging_product: str | None = None
    calls: list[dict[str, Any]] | None = None
    success: bool | None = None

    @property
    def call_id(self) -> str | None:
        if self.calls:
            first = self.calls[0]
            return first.get("id")
        return None

    contacts: list[MessageContact] = Field(default_factory=list)

    text/image/audio/video/document/sticker/location/contacts/reaction/

    timestamp: str | None = None
    type: str

    image: InboundMedia | None = None

    video: InboundMedia | None = None
