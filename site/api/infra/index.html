    Exceptions & HTTP¶

    wapi.exceptions ¶

    Exception hierarchy for the WhatsApp Business Platform.

    Every non-2xx response from the Graph API raises an :class:ApiError subclass. The subclass is chosen from the HTTP status and Meta's error code so callers can catch at whatever granularity they need::

    try:
    client.messages.send_text(to, "hi")
except RateLimitError as e:
    time.sleep(e.retry_after or 1)
except ApiError as e:
    log.error("Meta error %s (%s): %s", e.code, e.error_type, e)

    def _pick_class(code: int | None, status: int) -> type[ApiError]:
    if status == 401 or code == 190:
        return AuthError
    if status == 403 or code in (10, 200):
        return PermissionError_
    if status == 404 or (code == 100 and status == 404):
        return NotFoundError
    if code in _RATE_LIMIT_CODES:
        return RateLimitError
    if code is not None:
        if 131000 <= code < 132000:
            if code in (131051, 131052):
                return MediaError
            return MessagingError
        if 132000 <= code < 133000:
            return ValidationError
        if 133000 <= code < 134000:
            return RegistrationError
        if 138000 <= code < 139000:
            return CallError
    if status == 429:
        return RateLimitError
    if status >= 500:
        return ApiError
    return ApiError

    def error_from_response(
    status: int, payload: dict[str, Any], retry_after: float | None
) -> ApiError:
    """Build the right :class:`ApiError` subclass from an error response body."""
    err = payload.get("error") or {}
    code = err.get("code")
    cls = _pick_class(code, status)
    data = err.get("error_data") or {}
    return cls(
        err.get("message") or f"HTTP {status}",
        code=code,
        subcode=err.get("error_subcode"),
        error_type=err.get("type"),
        details=data.get("details") or err.get("error_user_msg"),
        fbtrace_id=err.get("fbtrace_id"),
        status_code=status,
        retry_after=retry_after,
        raw=payload,
    )

    @dataclass
class RetryPolicy:
    """Exponential backoff with jitter for retryable failures."""

    max_retries: int = 3
    base_delay: float = 0.5
    max_delay: float = 30.0

    def delay_for(self, attempt: int, retry_after: float | None) -> float:
        if retry_after is not None:
            return min(float(retry_after), self.max_delay)
        backoff = min(self.base_delay * (2**attempt), self.max_delay)
        return float(backoff * (0.5 + random.random() * 0.5))

    @dataclass
class Files:
    """Helper to build httpx multipart ``files`` payloads from user input.

    Accepts a path (str/Path), raw bytes, or an open binary file object.
    """

    @staticmethod
    def open(file: Any, filename: str | None = None) -> tuple[str, Any, str | None]:
        from pathlib import Path

        if isinstance(file, (str, Path)):
            path = Path(file)
            return (path.name, path.read_bytes(), _guess_mime(path.name))
        if isinstance(file, bytes):
            return (filename or "file", file, _guess_mime(filename or ""))
        # file-like object
        name = filename or getattr(file, "name", "file") or "file"
        name = str(name).rsplit("/", 1)[-1]
        return (name, file, _guess_mime(name))

    def handle_response(response: httpx.Response) -> dict[str, Any]:
    """Decode a Graph API response or raise the mapped :class:`ApiError`."""
    if response.is_success:
        if not response.content:
            return {}
        try:
            body = response.json()
        except json.JSONDecodeError:
            body = {"raw": response.text}
        return body if isinstance(body, dict) else {"data": body}
    payload: dict[str, Any]
    try:
        parsed = response.json()
        payload = parsed if isinstance(parsed, dict) else {"data": parsed}
    except json.JSONDecodeError:
        payload = {"error": {"message": response.text or f"HTTP {response.status_code}"}}
    raise error_from_response(response.status_code, payload, _retry_after(response.headers))

    def request_with_retry(
    http: httpx.Client,
    *,
    url: str,
    req: ApiRequest[T],
    headers: dict[str, str],
    retry: RetryPolicy,
) -> httpx.Response:
    """Execute one request, retrying retryable failures with backoff."""
    last_exc: ApiError | None = None
    for attempt in range(retry.max_retries + 1):
        response = http.request(
            req.method,
            url,
            params=req.params,
            json=req.json,
            data=req.data,
            files=req.files,
            headers=headers | (req.headers or {}),
        )
        if response.is_success:
            return response
        try:
            raise error_from_response(
                response.status_code,
                _safe_json(response),
                _retry_after(response.headers),
            )
        except ApiError as exc:
            last_exc = exc
            if attempt >= retry.max_retries or not is_retryable(exc):
                raise
            sleep(retry.delay_for(attempt, exc.retry_after))
    assert last_exc is not None  # unreachable
    raise last_exc

    def clean_body(body: dict[str, Any]) -> dict[str, Any]:
    """Recursively drop ``None`` values from a JSON body."""
    out: dict[str, Any] = {}
    for key, value in body.items():
        if value is None:
            continue
        if isinstance(value, dict):
            out[key] = clean_body(value)
        elif isinstance(value, list):
            out[key] = [clean_body(v) if isinstance(v, dict) else v for v in value]
        else:
            out[key] = value
    return out

    class CursorIterator(Generic[T], Iterator[T]):
    """Lazy iterator over a cursor-paginated endpoint."""

    def __init__(
        self,
        fetch: Fetcher,
        *,
        params: dict[str, Any] | None = None,
        decode: ItemDecoder[T] | None = None,
    ) -> None:
        self._fetch = fetch
        self._params = dict(params or {})
        self._decode: ItemDecoder[T] = decode or _passthrough
        self._buffer: list[T] = []
        self._next_url: str | None = None
        self._first = True
        self._exhausted = False
        self._seen_urls: set[str] = set()

    def __iter__(self) -> CursorIterator[T]:
        return self

    def __next__(self) -> T:
        while not self._buffer:
            if self._exhausted:
                raise StopIteration
            self._load_page()
        return self._buffer.pop(0)

    def _load_page(self) -> None:
        if self._first:
            payload = self._fetch(self._params, None)
            self._first = False
        else:
            assert self._next_url is not None
            payload = self._fetch(None, self._next_url)
        _, _, nxt = _cursors(payload)
        self._buffer.extend(self._decode(item) for item in payload.get("data") or [])
        # Guard against a server/mock returning the same cursor twice.
        if nxt and nxt not in self._seen_urls:
            self._next_url = nxt
            self._seen_urls.add(nxt)
        else:
            self._exhausted = True

    def pages(self) -> Iterator[Page[T]]:
        """Iterate page-by-page instead of item-by-item."""
        params: dict[str, Any] | None = self._params
        url: str | None = None
        first = True
        while True:
            payload = self._fetch(params if first else None, url)
            first = False
            before, after, nxt = _cursors(payload)
            items = [self._decode(i) for i in payload.get("data") or []]
            yield Page(data=items, before=before, after=after, has_next=bool(nxt))
            if not nxt:
                return
            url = nxt
